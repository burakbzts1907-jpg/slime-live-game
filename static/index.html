import os import asyncio import json from fastapi import FastAPI, WebSocket, WebSocketDisconnect from fastapi.staticfiles import StaticFiles from TikTokLive import TikTokLiveClient from TikTokLive.events import ConnectEvent, LikeEvent, GiftEvent app = FastAPI() # HTML dosyaları için static klasörünü bağlıyoruz app.mount("/static", StaticFiles(directory="static"), name="static") clients = set() # TikTok kullanıcı adını render.yaml içinden alıyoruz USERNAME = os.getenv("TIKTOK_USERNAME", "user27612891182928372810") tiktok = TikTokLiveClient(unique_id=USERNAME) async def broadcast(data): dead = [] for ws in clients: try: await ws.send_text(json.dumps(data)) except Exception: dead.append(ws) for ws in dead: clients.remove(ws) @tiktok.on(ConnectEvent) async def on_connect(event: ConnectEvent): await broadcast({"type": "log", "message": f"TikTok yayınına bağlanıldı: @{event.unique_id}"}) @tiktok.on(LikeEvent) async def on_like(event: LikeEvent): await broadcast({"type": "like", "message": f"{event.user.nickname} {event.likes} beğeni gönderdi!", "count": event.likes}) @tiktok.on(GiftEvent) async def on_gift(event: GiftEvent): if event.gift.streakable and event.gift.repeat_count > 1: msg = f"{event.user.nickname} {event.repeat_count} adet {event.gift.name} gönderdi!" else: msg = f"{event.user.nickname} {event.gift.name} gönderdi!" await broadcast({"type": "gift", "message": msg, "gift": event.gift.name}) @app.websocket("/ws") async def websocket_endpoint(websocket: WebSocket): await websocket.accept() clients.add(websocket) try: while True: data = await websocket.receive_text() packet = json.loads(data) action = packet.get("action") # Test butonları için simülasyon yanıtı if action == "tutkal": await broadcast({"type": "tutkal", "message": "Test: Beyaz tutkal eklendi! Şlop ses efekti..."}) elif action == "su": await broadcast({"type": "su", "message": "Test: Su eklendi! Şaaap ses efekti..."}) elif action == "ketchup": await broadcast({"type": "ketchup", "message": "Test: Ketçap eklendi! Slime rengi değişti."}) elif action == "like": await broadcast({"type": "like", "message": "Test: 100 Beğeni simüle edildi! Büyük su sıçraması."}) else: await broadcast({"type": "message", "message": f"Bilinmeyen eylem: {action}"}) except WebSocketDisconnect: clients.remove(websocket) @app.get("/") async def root(): from fastapi.responses import FileResponse return FileResponse("static/index.html") # TikTok istemcisini arka planda başlatma denemesi @app.on_event("startup") async def startup_event(): asyncio.create_task(tiktok.connect())
